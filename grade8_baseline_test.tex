\subsection*{Section A: Grade 7 Recap}

\textbf{Question 1: Integers and Operations [3]}
\begin{enumerate}[label=(\alph*)]
    \item $-7 + 15 - (-4)$ \\
    $= -7 + 15 + 4$ \checkmark \\
    $= 12$ \checkmark\checkmark
    \vspace{0.3cm}
    
    \item $(-3) \times 4 \times (-2)$ \\
    $= -12 \times (-2)$ \checkmark \\
    $= 24$ \checkmark
\end{enumerate}

\textbf{Question 2: Fractions and Percentages [3]}
\begin{enumerate}[label=(\alph*)]
    \item $\dfrac{2}{3} + \dfrac{3}{4}$ \\
    $= \dfrac{8}{12} + \dfrac{9}{12}$ \checkmark \\
    $= \dfrac{17}{12}$ or $1\dfrac{5}{12}$ \checkmark\checkmark
    \vspace{0.3cm}
    
    \item $0.35 = 35\%$ \checkmark \quad and \quad $\dfrac{7}{20} = 0.35$ \checkmark
\end{enumerate}

\textbf{Question 3: Basic Algebra [2]}
\begin{enumerate}[label=(\alph*)]
    \item $5x + 3y - 2x + 7y = 3x + 10y$ \checkmark
    \vspace{0.3cm}
    
    \item $3x + 7 = 22$ \\
    $3x = 15$ \checkmark \\
    $x = 5$ \checkmark
\end{enumerate}

\subsection*{Section B: Grade 8 Content}

\textbf{Question 4: Exponents and Powers [3]}
\begin{enumerate}[label=(\alph*)]
    \item $2^3 \times 2^4 = 2^{3+4} = 2^7$ \checkmark
    \vspace{0.3cm}
    
    \item $\dfrac{x^8}{x^3} = x^{8-3} = x^5$ \checkmark
    \vspace{0.3cm}
    
    \item $3\,500\,000 = 3.5 \times 10^6$ \checkmark
\end{enumerate}

\textbf{Question 5: Algebraic Expressions and Equations [3]}
\begin{enumerate}[label=(\alph*)]
    \item $3(2x - 5) + 4(x + 2)$ \\
    $= 6x - 15 + 4x + 8$ \checkmark\checkmark \\
    $= 10x - 7$ \checkmark
    \vspace{0.3cm}
    
    \item $2(x + 3) = 14$ \\
    $x + 3 = 7$ \checkmark \\
    $x = 4$ \checkmark
\end{enumerate}

\textbf{Question 6: Geometry [3]}
\begin{enumerate}[label=(\alph*)]
    \item $c^2 = a^2 + b^2$ \\
    $c^2 = 6^2 + 8^2$ \checkmark \\
    $c^2 = 36 + 64 = 100$ \\
    $c = 10$ cm \checkmark\checkmark
    \vspace{0.3cm}
    
    \item $A = \pi r^2 = \dfrac{22}{7} \times 7^2 = \dfrac{22}{7} \times 49 = 154$ cm$^2$ \checkmark
\end{enumerate}

\textbf{Question 7: Number Patterns [3]}
\begin{enumerate}[label=(\alph*)]
    \item Common difference $d = 4$ \checkmark
    \vspace{0.3cm}
    
    \item $T_{10} = 5 + (10-1) \times 4 = 5 + 36 = 41$ \checkmark
    \vspace{0.3cm}
    
    \item $T_n = 5 + (n-1) \times 4 = 4n + 1$ \checkmark
\end{enumerate}
